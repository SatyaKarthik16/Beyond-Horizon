

1. **Find the names of travelers who have booked a voyage departing in the next month.**

SELECT DISTINCT Traveler_name
FROM Traveler
JOIN Booking ON Traveler.Traveler_id = Booking.Traveler_id
JOIN Voyage_Info ON Booking.Voyage_id = Voyage_Info.Voyage_id
WHERE Voyage_Info.Departure_date BETWEEN GETDATE() AND DATEADD(month, 1, GETDATE());


   
2. **Retrieve the packages booked by travelers from a specific country.**

SELECT DISTINCT Package_name
FROM Package
JOIN Booking ON Package.Package_id = Booking.Package_id
JOIN Traveler ON Booking.Traveler_id = Traveler.Traveler_id
WHERE Traveler.Country = 'YourCountry';


3. **List the voyages with the highest number of bookings.**

SELECT Voyage_id, COUNT(*) AS Booking_count
FROM Booking
GROUP BY Voyage_id
ORDER BY Booking_count DESC
LIMIT 1;


4. **Find the average cost of bookings for each package.**

SELECT Package_name, AVG(Booking_cost) AS Average_cost
FROM Package
JOIN Booking ON Package.Package_id = Booking.Package_id
GROUP BY Package_name;


5. **Get the names of employees who have not been assigned any certifications.**


SELECT Employee_name
FROM Employees
WHERE Employee_id NOT IN (SELECT DISTINCT Employee_id FROM Certifications);


6. **Retrieve the names of travelers who have booked voyages with a spacecraft capacity greater than 10.**

SELECT DISTINCT Traveler_name
FROM Traveler
JOIN Booking ON Traveler.Traveler_id = Booking.Traveler_id
JOIN Voyage_Info ON Booking.Voyage_id = Voyage_Info.Voyage_id
JOIN Spacecraft ON Voyage_Info.Spacecraft_id = Spacecraft.Spacecraft_id
WHERE Spacecraft.Capacity > 10;


7. **List the voyages where the total payment amount exceeds the average payment amount across all voyages.**

SELECT Voyage_id, SUM(Payment_amount) AS Total_payment
FROM Payments
GROUP BY Voyage_id
HAVING Total_payment > (SELECT AVG(Payment_amount) FROM Payments);


8. **Find the total revenue generated from voyages in the last quarter.**


SELECT SUM(Payment_amount) AS Total_revenue
FROM Payments
WHERE Payment_date BETWEEN DATEADD(quarter, -1, GETDATE()) AND GETDATE();


9. **Retrieve the names of travelers who have made payments using a specific payment method.**

SELECT DISTINCT Traveler_name
FROM Traveler
JOIN Booking ON Traveler.Traveler_id = Booking.Traveler_id
JOIN Payments ON Booking.Booking_id = Payments.Booking_id
WHERE Payments.Payment_method = 'YourPaymentMethod';


10. **List the voyages that have crew members with certifications in more than one category.**

SELECT DISTINCT Voyage_id
FROM Crew
JOIN Certifications ON Crew.Certification_id = Certifications.Certification_id
GROUP BY Voyage_id
HAVING COUNT(DISTINCT Certifications.Category) > 1;


11. **Find the names of travelers who have booked the same package multiple times.**


SELECT DISTINCT Traveler_name
FROM Traveler
JOIN Booking ON Traveler.Traveler_id = Booking.Traveler_id
GROUP BY Traveler_name, Booking.Package_id
HAVING COUNT(*) > 1;

12. **Retrieve the voyages that have crew members with a specific role.**

SELECT DISTINCT Voyage_id
FROM Crew
WHERE Role = 'YourRole';


13. **List the packages that have not been booked by any traveler.**

SELECT Package_name
FROM Package
WHERE Package_id NOT IN (SELECT DISTINCT Package_id FROM Booking);


14. **Find the voyages where the average rating in reviews is greater than 4.**

SELECT DISTINCT Voyage_id
FROM Review
GROUP BY Voyage_id
HAVING AVG(Rating) > 4;


15. **Retrieve the names of employees who have booked voyages and also have certifications.**

SELECT DISTINCT Employee_name
FROM Employees
WHERE Employee_id IN (
    SELECT DISTINCT Employee_id
    FROM Certifications
    JOIN Crew ON Certifications.Certification_id = Crew.Certification_id
);


16. **List the voyages where the number of crew members is greater than the average number of crew members across all voyages.**

SELECT Voyage_id
FROM Crew
GROUP BY Voyage_id
HAVING COUNT(*) > (SELECT AVG(COUNT(*)) FROM Crew GROUP BY Voyage_id);


17. **Find the names of travelers who have made payments using the most common payment method.**


SELECT DISTINCT Traveler_name
FROM Traveler
JOIN Booking ON Traveler.Traveler_id = Booking.Traveler_id
JOIN Payments ON Booking.Booking_id = Payments.Booking_id
GROUP BY Traveler_name
ORDER BY COUNT(DISTINCT Payments.Payment_method) DESC
LIMIT 1;


18. **Retrieve the voyages that have crew members with certifications in a specific category.**

SELECT DISTINCT Voyage_id
FROM Crew
JOIN Certifications ON Crew.Certification_id = Certifications.Certification_id
WHERE Certifications.Category = 'YourCategory';


19. **List the voyages with the highest total payment amount.**

SELECT Voyage_id, SUM(Payment_amount) AS Total_payment
FROM Payments
GROUP BY Voyage_id
ORDER BY Total_payment DESC
LIMIT 1;


20. **Find the names of employees who have booked voyages with a departure date in the past month.**


SELECT DISTINCT Employee_name
FROM Employees
JOIN Booking ON Employees.Employee_id = Booking.Employee_id
JOIN Voyage_Info ON Booking.Voyage_id = Voyage_Info.Voyage_id
WHERE Voyage_Info.Departure_date BETWEEN DATEADD(month, -1, GETDATE()) AND GETDATE();
